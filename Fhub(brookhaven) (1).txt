local function createWeldConstraint(firstChild)
	local humanoidRootPartContainer = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	local Humanoid = humanoidRootPartContainer:FindFirstChildOfClass("Humanoid")

	if firstChild.Character and firstChild.Character:FindFirstChild("Head") then
		local Head = firstChild.Character.Head
		local HumanoidRootPart = humanoidRootPartContainer:FindFirstChild("HumanoidRootPart")

		if HumanoidRootPart then
			HumanoidRootPart.CFrame = Head.CFrame * CFrame.new(0, 2.2, 0)

			for _, child in pairs(HumanoidRootPart:GetChildren()) do
				if child:IsA("WeldConstraint") then
					child:Destroy()
				end
			end

			local WeldConstraint = Instance.new("WeldConstraint")

			WeldConstraint.Part0 = HumanoidRootPart
			WeldConstraint.Part1 = Head
			WeldConstraint.Parent = HumanoidRootPart

			if Humanoid then
				Humanoid.Sit = true
			end

			print("Headsit activated on " .. firstChild.Name)

			return true
		end

		warn("Your Character Has No Humanoid.")

		return false
	end

	warn("Target Doesn't Have Head Or Character.")

	return false
end
local function updateSit()
	local humanoidRootPartContainer = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	local Humanoid = humanoidRootPartContainer:FindFirstChildOfClass("Humanoid")
	local HumanoidRootPart = humanoidRootPartContainer:FindFirstChild("HumanoidRootPart")

	if HumanoidRootPart then
		for _, child in pairs(HumanoidRootPart:GetChildren()) do
			if child:IsA("WeldConstraint") then
				child:Destroy()
			end
		end
	end

	if Humanoid then
		Humanoid.Sit = false
	end

	print("Headsit deactivated")
end
local function handleInstance(instanceArgument)
	local lower = instanceArgument:lower()

	for _, player in pairs(players:GetPlayers()) do
		if player ~= LocalPlayer and lower == player.Name:lower():sub(1, #lower) then
			return player
		end
	end

	return nil
end

	Enabled = false,
	Connections = {},

local function updateHeartbeat(updateHeartbeatCondition)
	if updateHeartbeatCondition then
		for _, descendant in ipairs(updateHeartbeatCondition:GetDescendants()) do
			if descendant:IsA("BasePart") then
				descendant.CanCollide = not childData.Enabled
				descendant.Anchored = false
			end
		end

		return
	end
end
local function updateInstanceProperties(input)
	if input.Position.Y < -500 then
		local cFrame = CFrame.new(0, 100, 0)
		local raycastParams = RaycastParams.new()

		raycastParams.FilterDescendantsInstances = { localValuePlayer.Character }

		local raycastResult = Workspace:Raycast(input.Position, Vector3.new(0, 500, 0), raycastParams)

		input.CFrame = raycastResult and CFrame.new(raycastResult.Position + Vector3.new(0, 5, 0)) or cFrame
	end
end
local function secondaryUpdateHeartbeat()
	local updateHeartbeatCondition = Workspace:FindFirstChild("Com", true) and Workspace.Com:FindFirstChild("001_SoccerBalls")

	if updateHeartbeatCondition then
		for _, child in ipairs(updateHeartbeatCondition:GetChildren()) do
			if child.Name:match("^Soccer") then
				pcall(function()
					child.CanCollide = not childData.Enabled
					child.Anchored = childData.Enabled
				end)
				childData.SoccerBalls[child] = true
			end
		end

		if not childData.Connections.BallAdded then
			childData.Connections.BallAdded = updateHeartbeatCondition.ChildAdded:Connect(function(child)
				if child.Name:match("^Soccer") then
					task.wait(0.3)
					pcall(function()
						child.CanCollide = not childData.Enabled
						child.Anchored = childData.Enabled
					end)
				end
			end)
		end
	end
end
local function alternateUpdateHeartbeat()
	childData.Connections.Heartbeat = RunService.Heartbeat:Connect(function()
		local Character = localValuePlayer.Character

		if Character then
			updateHeartbeat(Character)

			local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

			if HumanoidRootPart then
				updateInstanceProperties(HumanoidRootPart)
			end
		end

		if tick() % 2 < 0.1 then
			secondaryUpdateHeartbeat()
		end
	end)
end

			end

			connection = localValuePlayer.CharacterAdded:Connect(function(character)
				if secondaryFlag then
					if character:WaitForChild("Humanoid", 5) then
						updateSeatedConnection(character)
					end

					return
				end

				connection:Disconnect()
			end)
		end
	end

local function new()
	if newText ~= "RGB" then
		if newText ~= "Preto" then
			if newText ~= "Branco" then
				if newText ~= "Vermelho" then
					if newText ~= "Verde" then
						if newText ~= "Azul" then
							if newText ~= "Amarelo" then
								if newText ~= "Rosa" then
									if newText ~= "Roxo" then
										return Color3.new(1, 1, 1)
									end

									return Color3.fromRGB(128, 0, 128)
								end

								return Color3.fromRGB(255, 105, 180)
							end

							return Color3.fromRGB(255, 255, 0)
						end

						return Color3.fromRGB(0, 170, 255)
					end

					return Color3.fromRGB(0, 255, 0)
				end

				return Color3.fromRGB(255, 0, 0)
			end

			return Color3.fromRGB(255, 255, 255)
		end

		return Color3.fromRGB(0, 0, 0)
	end

	local quotient = tick() % 5 / 5

	return Color3.fromHSV(quotient, 1, 1)
end
local function createEspBillboard(secondaryPlayer)
	if secondaryPlayer ~= alternatePlayers.LocalPlayer then
		if createEspBillboardFlag then
			local Character = secondaryPlayer.Character

			if Character then
				local Head = Character:FindFirstChild("Head")

				if Head then
					if createEspBillboardData[secondaryPlayer] then
						createEspBillboardData[secondaryPlayer]:Destroy()
					end

					local BillboardGui = Instance.new("BillboardGui")

					BillboardGui.Name = "ESP_Billboard"
					BillboardGui.Parent = Head
					BillboardGui.Adornee = Head
					BillboardGui.Size = UDim2.new(0, 200, 0, 50)
					BillboardGui.StudsOffset = Vector3.new(0, 3, 0)
					BillboardGui.AlwaysOnTop = true

					local TextLabel = Instance.new("TextLabel")

					TextLabel.Name = "TextLabel"
					TextLabel.Parent = BillboardGui
					TextLabel.Size = UDim2.new(1, 0, 1, 0)
					TextLabel.BackgroundTransparency = 1
					TextLabel.TextStrokeTransparency = 0.5
					TextLabel.Font = Enum.Font.SourceSansBold
					TextLabel.TextSize = 14
					TextLabel.Text = secondaryPlayer.Name .. " | " .. secondaryPlayer.AccountAge .. " dias"
					TextLabel.TextColor3 = new()
					createEspBillboardData[secondaryPlayer] = BillboardGui
				end
			end

			return
		end

		return
	end
end
local function cleanup(cleanupArgument)
	if createEspBillboardData[cleanupArgument] then
		createEspBillboardData[cleanupArgument]:Destroy()
		createEspBillboardData[cleanupArgument] = nil
	end
end

					end
				end
			end
		end)

		table.insert(connections, connection)

			createEspBillboard(player)

	end

	ToggleFallDamage = function(self, flag)
		if self.LocalPlayer.Character and self.LocalPlayer.Character:FindFirstChild("Humanoid") then
			local Humanoid = self.LocalPlayer.Character.Humanoid

			if not flag then
				Humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
				Humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)

				return false
			end

			Humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
			Humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
			Humanoid:SetStateEnabled(Enum.HumanoidStateType.GettingUp, true)
			Humanoid.PlatformStand = false

			return true
		end

		return false

	TeleportToSeat = function(self, instance, _)
		if self.LocalPlayer.Character and self.LocalPlayer.Character:FindFirstChild("Humanoid") then
			local Humanoid = self.LocalPlayer.Character.Humanoid
			local HumanoidRootPart = self.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

			if HumanoidRootPart then
				Humanoid.Sit = false
				task.wait(0.1)
				HumanoidRootPart.CFrame = instance.CFrame + Vector3.new(0, 5, 0)
				task.wait(0.1)
				instance:Sit(Humanoid)
				task.wait(0.5)

				return instance == Humanoid.SeatPart
			end

			self:MostrarNotificacao("Parte raiz do personagem não encontrada!")

			return false
		end

		self:MostrarNotificacao("Personagem não encontrado!")

		return false
	end,
	TeleportToVoid = function(self, bodyContainer)
		if bodyContainer then
			if not bodyContainer.PrimaryPart then
				local primaryPart = bodyContainer:FindFirstChild("Body", true) or bodyContainer:FindFirstChild("Chassis", true)

				if not primaryPart or not primaryPart:IsA("BasePart") then
					self:MostrarNotificacao("Parte principal do veículo não encontrada!")

					return
				end

				bodyContainer.PrimaryPart = primaryPart
			end

			local result = Vector3.new(0, -1000, 0)

			bodyContainer:SetPrimaryPartCFrame(CFrame.new(result))
			task.wait(0.5)

			return
		end

		self:MostrarNotificacao("Veículo inválido!")
	end,
	TeleportToPlayer = function(self, bodyContainer, sumNumber)
		if bodyContainer then
			if not bodyContainer.PrimaryPart then
				local primaryPart = bodyContainer:FindFirstChild("Body", true) or bodyContainer:FindFirstChild("Chassis", true)

				if not primaryPart or not primaryPart:IsA("BasePart") then
					self:MostrarNotificacao("Parte principal do veículo não encontrada!")

					return
				end

				bodyContainer.PrimaryPart = primaryPart
			end

			local sum = sumNumber + Vector3.new(5, 0, 5)

			bodyContainer:SetPrimaryPartCFrame(CFrame.new(sum))
			task.wait(0.5)

			return
		end

		self:MostrarNotificacao("Veículo inválido!")
	end,
	ExitCarAndReturn = function(self, condition)
		if self.LocalPlayer.Character and self.LocalPlayer.Character:FindFirstChild("Humanoid") then
			local Humanoid = self.LocalPlayer.Character.Humanoid

			if Humanoid.SeatPart then
				Humanoid.Sit = false
			end

			task.wait(0.1)

			if condition then
				self.LocalPlayer.Character:PivotTo(CFrame.new(condition))
			end

			return
		end
	end,
	AtualizarListaCarros = function(self)
		local Vehicles = self.Workspace:FindFirstChild("Vehicles")

			for _, child in ipairs(Vehicles:GetChildren()) do
				if child.Name:match("Car$") then
					table.insert(childNames, child.Name)
				end
			end
		end

		return childNames
	end

local function createSound(name, data, default)
	local names = {}
	local createSoundData = {}

	for k, item in pairs(data) do
		for _, item in ipairs(item) do
			if item.name ~= "" and item.id ~= "4354908569" then
				table.insert(names, item.name)
				createSoundData[item.name] = {
					id = item.id,
					category = k
				}
			end
		end
	end

	local id = nil
	local volume = 1
	local pitch = 1

	local function createSound(createSoundText, volume, pitch)
		sendRe("1Gu1nSound1s", {
			parent,
			createSoundText,
			volume
		})

		local Sound = Instance.new("Sound")

		Sound.Parent = parent
		Sound.SoundId = "rbxassetid://" .. createSoundText
		Sound.Volume = volume
		Sound.Pitch = pitch
		Sound.Looped = false
		Sound:Play()
		task.spawn(function()
			task.wait(3)
			Sound:Destroy()
		end)
	end

	nestedTab:AddDropdown({
		Name = name,
		Description = "Choose a sound to play on the server",
		Default = default,
		Multi = false,
		Options = names,
		Callback = function(createSoundFlag)
			if not createSoundFlag or not createSoundData[createSoundFlag] then
				id = nil
			else
				id = createSoundData[createSoundFlag].id
			end
		end
	})
	nestedTab:AddButton({
		Name = "Play Selected",
		Description = "Click to play the sound from the dropdown",
		Callback = function()
			if id then
				createSound(id, volume, pitch)
			end
		end
	})

	local createSoundFlag = false

	nestedTab:AddToggle({
		Name = "Loop",
		Description = "Loop Play Sound",
		Default = false,
		Callback = function(flag)
			createSoundFlag = flag

			if flag then
				task.spawn(function()
					while createSoundFlag do
						if id then
							createSound(id, volume, pitch)
						end

						task.wait(1)
					end
				end)
			end
		end
	})
end

local function sendPlayerToolEvent(argument)
	local ReplicatedStorage = game:GetService("ReplicatedStorage")
	local data = {
		[1] = "ToolMusicText",
		[2] = argument
	}

	ReplicatedStorage:WaitForChild("RE"):WaitForChild("PlayerToolEvent"):FireServer(unpack(data))

	local secondaryData = {
		[1] = "PickHouseMusicText",
		[2] = argument
	}

	ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Player1sHous1e"):FireServer(unpack(secondaryData))

	local alternateData = {
		[1] = "PickingCarMusicText",
		[2] = argument
	}

	ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Player1sCa1r"):FireServer(unpack(alternateData))

	local additionalData = {
		[1] = "PickingScooterMusicText",
		[2] = argument
	}

	ReplicatedStorage:WaitForChild("RE"):WaitForChild("1NoMoto1rVehicle1s"):FireServer(unpack(additionalData))
end

innerTab:AddTextBox({
	Name = "ID music",
	PlaceholderText = "Enter ID and press Enter",
	Callback = function(displayValue)
		if displayValue and displayValue ~= "" then
			sendPlayerToolEvent(tostring(displayValue))
		end
	end
})

local function handler(name, data, default)
	local names = {}
	local item = {}

	for k, secondaryItem in pairs(data) do
		for _, secondaryItem in ipairs(secondaryItem) do
			if secondaryItem.name ~= "" then
				table.insert(names, secondaryItem.name)
				item[secondaryItem.name] = {
					id = secondaryItem.id,
					category = k
				}
			end
		end
	end

	local function handler(displayValue)
		sendPlayerToolEvent(tostring(displayValue))
	end

	innerTab:AddDropdown({
		Name = name,
		Description = "all",
		Default = default,
		Multi = false,
		Options = names,
		Callback = function(idFlag)
			if idFlag and item[idFlag] then
				local id = item[idFlag].id

				if id and id ~= "" and id ~= "4354908569" then
					handler(id)
				end
			end
		end
	})
end

local function secondaryHandler(addDropdown, condition)
	pcall(function()

local function updateHeartbeatConnection(updateHeartbeatConnectionFlag)
	if additionalConnection then
		additionalConnection:Disconnect()
		additionalConnection = nil
	end

	local firstChild = innerPlayers:FindFirstChild(updateHeartbeatConnectionFlag)

	if not firstChild or firstChild == backpackContainer then
		updateCameraSubjectFlag = false
		capturedFirstChild = nil
	else
		capturedFirstChild = firstChild
		updateCameraSubjectFlag = true

		local function updateCameraSubject()
			if updateCameraSubjectFlag and capturedFirstChild then
				if not capturedFirstChild.Character or not capturedFirstChild.Character:FindFirstChild("Humanoid") then
					workspace.CurrentCamera.CameraSubject = nil
				else
					workspace.CurrentCamera.CameraSubject = capturedFirstChild.Character.Humanoid
				end

				return
			end
		end

		updateCameraSubject()
		additionalConnection = secondaryRunService.Heartbeat:Connect(function()
			if updateCameraSubjectFlag then
				pcall(updateCameraSubject)

				return
			end

			additionalConnection:Disconnect()
			additionalConnection = nil
		end)
		capturedFirstChild.CharacterAdded:Connect(function()
			if updateCameraSubjectFlag then
				updateCameraSubject()
			end
		end)
	end
end

local function updateTeleportToSelectedPlayer(text)
	local firstChild = innerPlayers:FindFirstChild(text)

	if not firstChild or not backpackContainer.Character or not backpackContainer.Character:FindFirstChild("HumanoidRootPart") or not firstChild.Character or not firstChild.Character:FindFirstChild("HumanoidRootPart") then
		print("Jogador ou personagem não encontrado para teletransporte.")
	else
		local HumanoidRootPart = backpackContainer.Character:FindFirstChild("HumanoidRootPart")
		local Humanoid = backpackContainer.Character:FindFirstChild("Humanoid")

		if not HumanoidRootPart or not Humanoid then
			print("Seu personagem não está totalmente carregado para teletransporte.")

			return
		end

		for _, descendant in ipairs(backpackContainer.Character:GetDescendants()) do
			if descendant:IsA("BasePart") then
				descendant.Velocity = Vector3.zero
				descendant.RotVelocity = Vector3.zero
				descendant.Anchored = true
			end
		end

		local ok, result = pcall(function()
			HumanoidRootPart.CFrame = CFrame.new(firstChild.Character.HumanoidRootPart.Position + Vector3.new(0, 2, 0))
		end)

		if not ok then
			warn("Erro ao teletransportar: " .. tostring(result))

			return
		end

		Humanoid.Sit = false
		Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
		task.wait(0.5)

		for _, descendant in ipairs(backpackContainer.Character:GetDescendants()) do
			if descendant:IsA("BasePart") then
				descendant.Anchored = false
				descendant.Velocity = Vector3.zero
				descendant.RotVelocity = Vector3.zero
			end
		end

		print("Teletransportado para o jogador: " .. text .. " com ancoragem segura.")
	end
end

local function handleSteppedConnection()
	local Backpack = backpackContainer:WaitForChild("Backpack")
	local couch = Backpack:FindFirstChild("Couch") or backpackContainer.Character:FindFirstChild("Couch")

	if not couch then
		local steppedConnectionData = {
			[1] = "PickingTools",
			[2] = "Couch"
		}

		if not pcall(function()
			ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Too1l"):InvokeServer(unpack(steppedConnectionData))
		end) then
			return false
		end

		repeat
			couch = Backpack:FindFirstChild("Couch")
			task.wait()
		until couch or task.wait(5)

		if not couch then
			return false
		end
	end

	if couch.Parent ~= backpackContainer.Character then
		couch.Parent = backpackContainer.Character
	end

	return true
end

local function updateSteppedConnection(updateSteppedConnectionFlag)
	if updateSteppedConnectionFlag and updateSteppedConnectionFlag.Character and backpackContainer.Character then
		local Character = backpackContainer.Character
		local Humanoid = Character:FindFirstChildOfClass("Humanoid")
		local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

		if Humanoid and HumanoidRootPart then
			HumanoidRootPartPosition = HumanoidRootPart.Position
			pcall(function()
				HumanoidRootPart.Anchored = true
				HumanoidRootPart.CFrame = CFrame.new(Vector3.new(1181.83, 76.08, -1158.83))
				task.wait(0.2)
				HumanoidRootPart.Velocity = Vector3.zero
				HumanoidRootPart.RotVelocity = Vector3.zero
				HumanoidRootPart.Anchored = false

				if Humanoid then
					Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
				end
			end)
			task.wait(0.5)
			updateSteppedConnectionDisabled()

			local updateSteppedConnectionData = {
				[1] = "DeleteAllVehicles"
			}

			pcall(function()
				ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Ca1r"):FireServer(unpack(updateSteppedConnectionData))
			end)
			updateSteppedConnectionData = {
				[1] = "PickingCar",
				[2] = "SchoolBus"
			}
			pcall(function()
				ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Ca1r"):FireServer(unpack(updateSteppedConnectionData))
			end)
			task.wait(1)

			local Vehicles = workspace:FindFirstChild("Vehicles")

			if Vehicles then
				local firstChild = Vehicles:FindFirstChild(backpackContainer.Name .. "Car")

				if firstChild then
					pcall(function()
						HumanoidRootPart.Anchored = true
						HumanoidRootPart.CFrame = CFrame.new(Vector3.new(1171.15, 79.45, -1166.2))
						task.wait(0.2)
						HumanoidRootPart.Velocity = Vector3.zero
						HumanoidRootPart.RotVelocity = Vector3.zero
						HumanoidRootPart.Anchored = false
						Humanoid:ChangeState(Enum.HumanoidStateType.Seated)
					end)

					local timestamp = tick()

					repeat
						task.wait()

						if tick() - timestamp > 10 then
							return
						end
					until Humanoid.Sit

					for _, descendant in ipairs(firstChild:GetDescendants()) do
						if descendant:IsA("BasePart") then
							descendant.CanCollide = false
							pcall(function()
								descendant:SetNetworkOwner(nil)
							end)
						end
					end

					createBodyVelocityFlag = true
					connection = secondaryRunService.Stepped:Connect(function()
						if createBodyVelocityFlag then
							for _, descendant in ipairs(Character:GetDescendants()) do
								if descendant:IsA("BasePart") then
									descendant.CanCollide = false
								end
							end

							return
						end
					end)

					local updateSteppedConnectionNumber = tick()
					local number = 0.05
					local secondaryUpdateSteppedConnectionNumber = tick()

					alternateConnection = secondaryRunService.Heartbeat:Connect(function()
						if createBodyVelocityFlag then
							local humanoidContainer = updateSteppedConnectionFlag.Character or updateSteppedConnectionFlag.CharacterAdded:Wait()
							local humanoidRootPart = humanoidContainer:FindFirstChild("HumanoidRootPart")
							local humanoid = humanoidContainer:FindFirstChild("Humanoid")

							if humanoidRootPart and humanoid then
								if HumanoidRootPart and Humanoid then
									if not (tick() - updateSteppedConnectionNumber < number) then
										updateSteppedConnectionNumber = tick()

										local sumNumber = Vector3.new(math.random(-10, 10), 0, math.random(-10, 10))

										pcall(function()
											local updateSteppedConnectionNumber = humanoidRootPart.Position + sumNumber

											firstChild:PivotTo(CFrame.new(updateSteppedConnectionNumber) * CFrame.Angles(math.rad(workspace.DistributedGameTime * 12000), math.rad(workspace.DistributedGameTime * 15000), math.rad(workspace.DistributedGameTime * 18000)))
										end)

										local updateSteppedConnectionFlag = false

										for _, descendant in ipairs(firstChild:GetDescendants()) do
											if (descendant:IsA("Seat") or descendant:IsA("VehicleSeat")) and descendant.Name ~= "VehicleSeat" and humanoid == descendant.Occupant then
												updateSteppedConnectionFlag = true

												break
											end
										end

										if updateSteppedConnectionFlag or tick() - secondaryUpdateSteppedConnectionNumber > 10 then
											createBodyVelocityFlag = false

											if connection then
												connection:Disconnect()
												connection = nil
											end

											if alternateConnection then
												alternateConnection:Disconnect()
												alternateConnection = nil
											end

											pcall(function()
												firstChild:PivotTo(CFrame.new(Vector3.new(-76.6, -401.97, -84.26)))
											end)
											task.wait(0.5)
											updateSteppedConnectionDisabled()

											local updateSteppedConnectionData = {
												[1] = "DeleteAllVehicles"
											}

											pcall(function()
												ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Ca1r"):FireServer(unpack(updateSteppedConnectionData))
											end)

											if Character then
												local humanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

												if humanoidRootPart and HumanoidRootPartPosition then
													pcall(function()
														humanoidRootPart.Anchored = true
														humanoidRootPart.CFrame = CFrame.new(HumanoidRootPartPosition + Vector3.new(0, 5, 0))
														task.wait(0.2)
														humanoidRootPart.Velocity = Vector3.zero
														humanoidRootPart.RotVelocity = Vector3.zero
														humanoidRootPart.Anchored = false

														if Humanoid then
															Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
														end
													end)
												end
											end

											if Character then
												for _, descendant in ipairs(Character:GetDescendants()) do
													if descendant:IsA("BasePart") then
														descendant.CanCollide = true
														descendant.Velocity = Vector3.zero
														descendant.RotVelocity = Vector3.zero
													end
												end
											end

											local updateSteppedConnectionCondition = Character and Character:FindFirstChild("Humanoid")

											if updateSteppedConnectionCondition then
												updateSteppedConnectionCondition:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
											end

											for _, descendant in ipairs(workspace:GetDescendants()) do
												if descendant:IsA("Seat") or descendant:IsA("VehicleSeat") then
													descendant.Disabled = false
												end
											end

											pcall(function()
												ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Clothe1s"):FireServer("CharacterSizeUp", 1)
											end)
										end

										return
									end

									return
								end

								createBodyVelocityFlag = false

								return
							end

							return
						end
					end)

					return
				end

				return
			end

			return
		end

		return
	end

local function handleBodyVelocity()
	local Backpack = backpackContainer:WaitForChild("Backpack")
	local soccerBall = Backpack:FindFirstChild("SoccerBall") or backpackContainer.Character:FindFirstChild("SoccerBall")

	if not soccerBall then
		local bodyVelocityData = {
			[1] = "PickingTools",
			[2] = "SoccerBall"
		}

		if not pcall(function()
			ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Too1l"):InvokeServer(unpack(bodyVelocityData))
		end) then
			return false
		end

		repeat
			soccerBall = Backpack:FindFirstChild("SoccerBall")
			task.wait()
		until soccerBall or task.wait(5)

		if not soccerBall then
			return false
		end
	end

	if soccerBall.Parent ~= backpackContainer.Character then
		soccerBall.Parent = backpackContainer.Character
	end

	return true
end
